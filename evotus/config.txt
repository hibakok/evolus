# -----------------------------------------------------------------------------
# РАЗМЕРЫ ПОПУЛЯЦИИ
# -----------------------------------------------------------------------------

# Количество особей во внутренней популяции (элита)
# Чем больше популяция, тем разнообразнее поиск решений, но медленнее работа.
# Рекомендуемые значения: 30-100 для простых задач, 50-200 для сложных.
# Для быстрого старта можно использовать 10-20.
population_size = 20

# Сколько потомков создаёт каждая особь внутренней популяции в каждом поколении
# Увеличение повышает шанс найти удачную мутацию, но замедляет работу.
# Рекомендуемые значения: 3-8
offspring_per_individual = 2

# -----------------------------------------------------------------------------
# ПАРАМЕТРЫ МУТАЦИЙ
# -----------------------------------------------------------------------------

# Базовое количество минимальных мутаций, применяемых к одному потомку
# внутренней популяции (прародителей). Каждая мутация может изменить вес,
# добавить/удалить связь или нейрон.
# Рекомендуемые значения: 3-10
mutations_per_offspring = 5

# Количество мутаций для особей внешней популяции (потомков, создаваемых от
# прародителей внутренней популяции). Позволяет настроить уровень изменчивости
# для исследуемых решений без изменения параметров внутренней популяции.
# Рекомендуемые значения: 3-10
outer_population_mutations = 5

# Вероятность того, что мутация изменит вес существующей связи (0.0 - 1.0)
# Сумма всех mutation_rate_* должна быть примерно равна 1.0
mutation_rate_weight = 0.3

# Вероятность того, что мутация добавит или удалит связь (0.0 - 1.0)
mutation_rate_connection = 0.2

# Вероятность того, что мутация добавит или удалит нейрон (0.0 - 1.0)
mutation_rate_neuron = 0.2

# Вероятность того, что мутация изменит функцию активации нейрона (0.0 - 1.0)
mutation_rate_activation = 0.3

# -----------------------------------------------------------------------------
# СИЛА МУТАЦИЙ (адаптируется автоматически во время эволюции)
# -----------------------------------------------------------------------------

# Начальное стандартное отклонение для мутации весов
# Определяет, насколько сильно могут измениться веса за одну мутацию.
# 0.5 означает изменение веса примерно на ±50% от текущего значения.
weight_mutation_std = 0.5

# Минимальная сила мутаций (для точной настройки near оптимума)
# Когда эволюция находит хорошее решение, сила мутаций уменьшается для тонкой настройки.
min_mutation_std = 0.01

# Максимальная сила мутаций (для выхода из локальных минимумов)
# Если прогресса нет долгое время, сила мутаций увеличивается для исследования.
max_mutation_std = 2.0

# -----------------------------------------------------------------------------
# СТРУКТУРА СЕТИ
# -----------------------------------------------------------------------------

# Количество скрытых нейронов при инициализации сети
# Больше нейронов дают больше возможностей, но увеличивают сложность.
# Рекомендуемые значения: 2-10 для начала
initial_hidden_neurons = 4

# =============================================================================
# СОВЕТЫ ПО НАСТРОЙКЕ
# =============================================================================
# 1. Если эволюция застревает (нет прогресса):
#    - Увеличьте population_size (больше разнообразия)
#    - Увеличьте offspring_per_individual (больше попыток мутаций)
#    - Увеличьте max_mutation_std (более смелые изменения)
#    - Увеличьте outer_population_mutations (большая изменчивость потомков)
#
# 2. Если эволюция слишком медленная:
#    - Уменьшите population_size
#    - Уменьшите offspring_per_individual
#    - Уменьшите outer_population_mutations (меньше мутаций на потомка)
#
# 3. Для сложных задач (например XOR, классификация):
#    - population_size = 50-100
#    - offspring_per_individual = 4-6
#    - mutations_per_offspring = 5-8
#    - outer_population_mutations = 5-10
#    - initial_hidden_neurons = 4-8
#
# 4. Для простых задач (линейная регрессия):
#    - population_size = 20-30
#    - offspring_per_individual = 2-4
#    - mutations_per_offspring = 2-4
#    - outer_population_mutations = 3-6
#    - initial_hidden_neurons = 2-4
# =============================================================================
