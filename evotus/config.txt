# ============================================
# Файл конфигурации Evotus
# ============================================
# В этом файле задаются все настройки эволюции нейросети.
# Каждая настройка имеет комментарий на русском языке.
# Изменяйте значения после знака '='. Не удаляйте сами настройки.
# ============================================

# Размер внутренней популяции (количество особей, которые хранятся постоянно)
# Чем больше популяция, тем разнообразнее поиск решений, но медленнее работа.
# Рекомендуемые значения: 20-100
population_size = 20

# Количество потомков, которое создаёт каждая особь внутренней популяции в каждом поколении
# Каждый потомок получает мутации и тестируется во внешней популяции.
# Рекомендуемые значения: 2-5
offspring_per_individual = 3

# Количество минимальных мутаций, применяемых к каждому потомку
# Мутации могут быть: изменение веса, добавление/удаление связи, 
# добавление/удаление нейрона, изменение функции активации.
# Рекомендуемые значения: 3-10
mutations_per_offspring = 5

# Вероятность того, что мутация будет изменением веса связи (от 0.0 до 1.0)
# Остальные типы мутаций распределяются между оставшейся вероятностью.
# Рекомендуемые значения: 0.2-0.4
mutation_rate_weight = 0.3

# Вероятность того, что мутация будет добавлением или удалением связи (от 0.0 до 1.0)
# Эта вероятность используется вместе с предыдущей для определения типа мутации.
# Рекомендуемые значения: 0.1-0.3
mutation_rate_connection = 0.2

# Вероятность того, что мутация будет добавлением или удалением нейрона (от 0.0 до 1.0)
# Рекомендуемые значения: 0.1-0.3
mutation_rate_neuron = 0.2

# Вероятность того, что мутация будет изменением функции активации нейрона (от 0.0 до 1.0)
# Это остаточная вероятность после учёта всех предыдущих типов мутаций.
# Рекомендуемые значения: 0.1-0.3
mutation_rate_activation = 0.3

# Стандартное отклонение для изменения веса при мутации веса
# Определяет силу изменения веса: чем больше значение, тем сильнее изменения.
# Рекомендуемые значения: 0.1-1.0
weight_mutation_std = 0.5
